**Step 3: Compare and build verification output**
Write a Python script that:
```python
import json
from pathlib import Path
from datetime import date

TOLERANCE = 0.001  # 0.10%
phase_dir = Path(".planning/phases/01-pre-implementation-audit")
cv_dir = Path("outputs/classifier_cv")

folds_result = {}
for fold_num in ["01", "02", "03", "04", "05"]:
    # Load reported metrics
    reported_path = cv_dir / f"fold_{fold_num}" / "test_results.json"
    with open(reported_path) as f:
        reported = json.load(f)
    reported_acc = reported["metrics"]["accuracy"]

    # Load verified metrics
    verified_path = phase_dir / f"verify_fold_{fold_num}.json"
    with open(verified_path) as f:
        verified = json.load(f)
    verified_acc = verified["metrics"]["accuracy"]

    # Compare
    diff = abs(verified_acc - reported_acc)
    folds_result[f"fold_{fold_num}"] = {
        "reported_accuracy": round(reported_acc, 6),
        "verified_accuracy": round(verified_acc, 6),
        "difference": round(diff, 6),
        "within_tolerance": diff < TOLERANCE
    }

# Compute aggregate
reported_mean = sum(f["reported_accuracy"] for f in folds_result.values()) / 5
verified_mean = sum(f["verified_accuracy"] for f in folds_result.values()) / 5
agg_diff = abs(verified_mean - reported_mean)

all_pass = all(f["within_tolerance"] for f in folds_result.values()) and agg_diff < TOLERANCE

output = {
    "verification_date": str(date.today()),
    "tolerance": TOLERANCE,
    "folds": folds_result,
    "aggregate": {
        "reported_mean": round(reported_mean, 6),
        "verified_mean": round(verified_mean, 6),
        "difference": round(agg_diff, 6),
        "within_tolerance": agg_diff < TOLERANCE
    },
    "status": "PASS" if all_pass else "FAIL"
}

with open(phase_dir / "BASELINE_VERIFICATION.json", "w") as f:
    json.dump(output, f, indent=2)

print(f"Status: {output['status']}")
```
